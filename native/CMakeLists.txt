cmake_minimum_required(VERSION 3.18)
project(ayzygisk_native VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ay-engine-core STATIC engine/core.cpp)
target_include_directories(ay-engine-core PUBLIC engine)
target_compile_options(ay-engine-core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -fstack-protector-strong)
add_library(ay-engine-recovery STATIC engine/recovery.cpp)
target_include_directories(ay-engine-recovery PUBLIC engine)
target_compile_options(ay-engine-recovery PRIVATE -Wall -Wextra -Wpedantic -Wconversion -fstack-protector-strong)

add_executable(ay-native src/main.cpp)
add_executable(ay-engine-inventory engine/inventory.cpp engine/inventory_main.cpp)
target_include_directories(ay-engine-inventory PRIVATE engine)
target_compile_options(ay-engine-inventory PRIVATE -Wall -Wextra -Wpedantic -Wconversion -fstack-protector-strong)
target_compile_options(ay-native PRIVATE -Wall -Wextra -Wpedantic -Wconversion -fstack-protector-strong)
if(ANDROID)
  # ONLY the safe read-only inspector is built on Android.
  target_link_options(ay-native PRIVATE -Wl,-z,relro,-z,now)
else()
  enable_testing()
  add_executable(ay-engine-core-test engine/tests/test_core.cpp)
  target_link_libraries(ay-engine-core-test PRIVATE ay-engine-core)
  target_compile_options(ay-engine-core-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME ay-engine-core COMMAND ay-engine-core-test)
  add_executable(ay-engine-recovery-test engine/tests/test_recovery.cpp)
  target_link_libraries(ay-engine-recovery-test PRIVATE ay-engine-recovery)
  target_compile_options(ay-engine-recovery-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME ay-engine-recovery COMMAND ay-engine-recovery-test)
  # Separate, isolated desktop-only lab for AY TEST plugins (not Zygisk modules).
  add_executable(ay-runtime-host runtime/host_runtime.cpp)
  target_include_directories(ay-runtime-host PRIVATE runtime)
  target_link_libraries(ay-runtime-host PRIVATE ${CMAKE_DL_LIBS})
  target_compile_options(ay-runtime-host PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
  foreach(x ok wrong crash hang)
    if(x STREQUAL "ok")
      set(source runtime/tests/plugin_ok.cpp)
    elseif(x STREQUAL "wrong")
      set(source runtime/tests/plugin_wrong_abi.cpp)
    elseif(x STREQUAL "crash")
      set(source runtime/tests/plugin_crash.cpp)
    else()
      set(source runtime/tests/plugin_hang.cpp)
    endif()
    add_library(ay-test-${x} SHARED ${source})
    target_include_directories(ay-test-${x} PRIVATE runtime)
    set_target_properties(ay-test-${x} PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
  endforeach()
endif()
