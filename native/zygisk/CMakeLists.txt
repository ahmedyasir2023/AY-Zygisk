cmake_minimum_required(VERSION 3.22)
project(ay_zygisk_bridge LANGUAGES CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "Zygisk ABI bridge must be built with the Android NDK toolchain")
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/zygisk.hpp")
    message(FATAL_ERROR "Official zygisk.hpp missing: see native/zygisk/README.md")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/vendor/zygisk.hpp" HEADER_PREFIX LIMIT 3500)
if(NOT HEADER_PREFIX MATCHES "#define ZYGISK_API_VERSION 5")
    message(FATAL_ERROR "Expected official Zygisk public API v5 header")
endif()
add_library(ay_zygisk_bridge SHARED ay_bridge.cpp)
target_include_directories(ay_zygisk_bridge PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor")
target_compile_features(ay_zygisk_bridge PRIVATE cxx_std_17)
target_compile_options(ay_zygisk_bridge PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti -fvisibility=hidden -ffunction-sections -fdata-sections)
target_link_options(ay_zygisk_bridge PRIVATE -Wl,--gc-sections -Wl,--no-undefined)
set_target_properties(ay_zygisk_bridge PROPERTIES OUTPUT_NAME "ayzygisk" PREFIX "lib")
